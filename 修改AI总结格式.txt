function extractJsonFromText(text) {
  if (!text) return null;

  // 去掉可能出现的 ```json 代码块
  let cleaned = text
    .replace(/```json/g, "")
    .replace(/```/g, "")
    .trim();

  // 如果 AI 前后多说了几句话，尽量截取第一个 { 到最后一个 }
  const firstBrace = cleaned.indexOf("{");
  const lastBrace = cleaned.lastIndexOf("}");

  if (firstBrace !== -1 && lastBrace !== -1 && lastBrace > firstBrace) {
    cleaned = cleaned.slice(firstBrace, lastBrace + 1);
  }

  return JSON.parse(cleaned);
}

function fallbackReview(rawText = "") {
  return {
    title: "AI 复盘生成中出现了一点小插曲",

# 三、修改 `server.js` 里的接口返回

```js
const result = completion.choices?.[0]?.message?.content || "AI 暂未返回内容。";
res.json({
  provider: AI_PROVIDER,
  model: provider.model,
  result

```

把它替换成：

```js
const rawText = completion.choices?.[0]?.message?.content || "";

try {
  review = extractJsonFromText(rawText);
} catch (parseError) {
  console.warn("AI 返回内容不是标准 JSON，使用兜底内容。");
  console.warn("AI 原始返回：", rawText);

  provider: AI_PROVIDER,

function renderAIReview(review, payload) {
  const listHtml = (items) => {
    if (!Array.isArray(items) || !items.length) return "<li>暂无</li>";
    return items.map(item => `<li>${escapeHtml(item)}</li>`).join("");
  };

  return `
    <h3>AI 个性化职场能力报告</h3>

    <div class="ai-review-meta">
      <p>
        AI 分析依据：
        分析力 ${payload.scores.analysis}，
        沟通力 ${payload.scores.communication}，
        方案完整度 ${payload.scores.plan}，
        选择记录 ${payload.choices.length} 条。
      </p>
    </div>

    <div class="ai-review-content">
      <h4>${escapeHtml(review.title || "职场能力复盘")}</h4>

      <p><strong>综合评价：</strong>${escapeHtml(review.scoreSummary || "")}</p>

      <p><strong>老板点评：</strong>${escapeHtml(review.bossComment || "")}</p>

      <h4>你的优势</h4>
      <ul>${listHtml(review.strengths)}</ul>

      <h4>还可以提升</h4>
      <ul>${listHtml(review.weaknesses)}</ul>

      <h4>适合尝试的实习方向</h4>
      <ul>${listHtml(review.jobFit)}</ul>

      <h4>AI 分析依据</h4>
      <ul>${listHtml(review.evidence)}</ul>

      <h4>7 天行动计划</h4>
      <ul>${listHtml(review.sevenDayPlan)}</ul>

      <p><strong>下一步建议：</strong>${escapeHtml(review.suggestion || "")}</p>
    </div>
  `;
}
```

# 最简单总结

你要改的位置是：

```text
server.js
1. 修改 buildReviewPrompt，让 AI 只返回 JSON
2. 新增 extractJsonFromText 和 fallbackReview
3. 修改 /api/ai-review 的 res.json，从 result 改成 review

index.html
1. 新增 renderAIReview(review, payload)
2. 修改 generateAIReview，把 data.result 改成 data.review
3. 补一点 CSS，让 JSON 字段分卡片展示
```

核心目标是把：

```text
AI 返回一整段文字
```

改成：

```text
AI 返回结构化 review 对象
前端按字段展示
```
